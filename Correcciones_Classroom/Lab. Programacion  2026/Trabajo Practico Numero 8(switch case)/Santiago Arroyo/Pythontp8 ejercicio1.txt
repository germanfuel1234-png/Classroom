canciones=int(input("Ingrese un numero de canciones... "))
partituras=int(input("Ingrese un numero de partituras... "))
if canciones>=7 and partituras==0:
    print(f"Eres un músico naciente")
if canciones>=7 and partituras>=1:
    print(f"Eres un músico estelar")
if canciones>=10 and partituras>=5:
    print(f"Eres un músico consagrado")
